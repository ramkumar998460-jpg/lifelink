Project Structure, Supabase SQL Schema & Multi-File Setup

Complete database DDL with Row Level Security (RLS), modular folder tree, and setup README.

-- LIFELINK SUPABASE POSTGRESQL SCHEMA & ROW LEVEL SECURITY (RLS)
-- Run this script in your Supabase SQL Editor

create extension if not exists "pgcrypto";

-- 1. PROFILES TABLE (Linked to auth.users)
create table public.profiles (
    id uuid references auth.users on delete cascade primary key,
    full_name text not null,
    email text unique not null,
    phone text,
    role text not null default 'user' check (role in ('user', 'admin')),
    created_at timestamptz not null default now()
);

-- 2. HOSPITALS TABLE (Sample Exhibition Directory)
create table public.hospitals (
    id uuid primary key default gen_random_uuid(),
    name text not null,
    area text not null,
    address text not null,
    phone text not null,
    category text not null,
    latitude double precision not null,
    longitude double precision not null,
    is_sample_record boolean not null default true,
    created_at timestamptz not null default now()
);

-- 3. DONORS TABLE (Privacy-Protected Blood Registry)
create table public.donors (
    id uuid primary key default gen_random_uuid(),
    user_id uuid references public.profiles(id) on delete set null,
    donor_code text unique not null,
    masked_name text not null,
    blood_group text not null check (blood_group in ('A+','A-','B+','B-','O+','O-','AB+','AB-')),
    city text not null default 'Mumbai',
    area text not null,
    private_phone text not null, -- Protected by RLS / view masking
    availability text not null default 'Available',
    consent_given boolean not null default true,
    created_at timestamptz not null default now()
);

-- 4. ASSISTANCE REQUESTS TABLE
create table public.assistance_requests (
    id uuid primary key default gen_random_uuid(),
    reference_id text unique not null,
    user_id uuid references public.profiles(id) on delete set null,
    full_name text not null,
    contact_info text not null,
    category text not null,
    location text not null,
    description text not null,
    status text not null default 'Pending Review' check (status in ('Pending Review', 'In Progress', 'Resolved')),
    created_at timestamptz not null default now()
);

-- ENABLE ROW LEVEL SECURITY (RLS)
alter table public.profiles enable row level security;
alter table public.hospitals enable row level security;
alter table public.donors enable row level security;
alter table public.assistance_requests enable row level security;

-- RLS POLICIES
-- Hospitals: Public read, Admin write
create policy "Public can view sample hospitals" on public.hospitals for select using (true);
create policy "Admins can manage hospitals" on public.hospitals for all using (
    exists (select 1 from public.profiles where id = auth.uid() and role = 'admin')
);

-- Assistance Requests: Users view own requests, Admins view/update all
create policy "Users view own requests" on public.assistance_requests for select using (
    auth.uid() = user_id or exists (select 1 from public.profiles where id = auth.uid() and role = 'admin')
);
create policy "Anyone can submit assistance request" on public.assistance_requests for insert with check (true);
create policy "Admins can update request status" on public.assistance_requests for update using (
    exists (select 1 from public.profiles where id = auth.uid() and role = 'admin')
);

# LifeLink – Emergency Assistance & Community Support Platform
**Tagline:** *Connecting Lives. Delivering Hope.*

## 1. Overview
LifeLink is a modern emergency assistance and community support platform built with HTML5, CSS3, Vanilla JavaScript, Supabase, Leaflet + OpenStreetMap, and Chart.js.

## 2. Ethical & Safety Guardrails
- **Not an Emergency Dispatch Service:** Clearly advises citizens to dial **112 / 108** for real emergencies.
- **Honest Hospital Directory:** Displays `[SAMPLE EXHIBITION RECORD]` badges and never fabricates live ICU/bed availability.
- **Strict Donor Privacy:** Masks personal phone numbers behind a consent-protected relay workflow.

## 3. Running in Exhibition Mode vs. Live Supabase Mode
1. **Exhibition Demo Mode (Default):** Works immediately out-of-the-box offline or online using pre-seeded synthetic records in `localStorage`.
2. **Live Supabase Mode:**
   - Create a project at https://supabase.com
   - Run `supabase_schema.sql` in the Supabase SQL Editor.
   - Click **DB: Exhibition LocalStorage** in the top bar and paste your Project URL and Anon Key.
